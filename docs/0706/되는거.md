// ==UserScript==
// @name         E2R ChatGPT Auto Sender기존2222
// @namespace    http://tampermonkey.net/
// @version      7.1
// @match        https://chatgpt.com/*
// @match        https://chat.openai.com/*
// @run-at       document-idle
// @grant        none
// ==/UserScript==

(function () {
  'use strict';

  const PROMPT = `
아래 GitHub Raw URL의 실행 프롬프트 전체를 열고 그대로 이행해.

[MAIN EXECUTION PROMPT]
https://raw.githubusercontent.com/Songdaiki/stock_agent/main/docs/core/e2r_v12_prompt_round_scheduler_corrected.txt

[NO-REPEAT INDEX]
https://raw.githubusercontent.com/Songdaiki/stock_agent/main/docs/core/V12_Research_No_Repeat_Index.md

MAIN EXECUTION PROMPT 전체가 이번 실행의 기본 연구 절차다.
NO-REPEAT INDEX는 중복 방지 장부로만 사용한다.`.trim();

  const FIRST_DELAY = 60 * 1000;
  const NEXT_DELAY = 3 * 60 * 1000;

  const HARVEST_POLL_MS = 5000;
  const DOWNLOAD_SETTLE_DELAY = 10 * 1000;

  // 일반적으로 md가 안 생긴 상태를 오래 기다릴 한계
  const NO_MD_RETRY_LIMIT = 60; // 60 * 5초 = 5분

  // 오류 문구가 보이면 훨씬 빨리 재전송
  const ERROR_NO_MD_RETRY_LIMIT = 2; // 2 * 5초 = 10초
  const ERROR_RECOVERY_DELAY = 20 * 1000;

  const CLICKED_KEY = 'E2R_CLICKED_MD_BUTTONS_SIMPLE_V71';

  let count = 0;
  let scheduled = false;

  // 전송 후 md 다운로드를 기다리는 중인지
  let awaitingMdDownload = false;

  // md 다운로드 시도 중인지
  let harvesting = false;

  // 이번 전송 전에 이미 화면에 있던 md 버튼들
  let preExistingMdKeys = new Set();

  let noMdRetries = 0;
  let autoPaused = false;
  let recoveryCount = 0;

  console.error('### E2R SIMPLE V7.1 LOADED ###');

  /******************************************************************
   * BASIC DOM
   ******************************************************************/

  function getEditor() {
    return document.querySelector('div.ProseMirror[contenteditable="true"]')
      || document.querySelector('[contenteditable="true"]')
      || document.querySelector('textarea[name="prompt-textarea"]')
      || document.querySelector('#prompt-textarea');
  }

  function getComposerForm() {
    const editor = getEditor();

    return editor?.closest('form')
      || document.querySelector('form[data-type="unified-composer"]')
      || document.querySelector('form');
  }

  function getSendButton() {
    const direct =
      document.querySelector('#composer-submit-button')
      || document.querySelector('button[data-testid="send-button"]')
      || document.querySelector('button[aria-label*="프롬프트 보내기"]')
      || document.querySelector('button[aria-label*="전송"]')
      || document.querySelector('button[aria-label*="Send"]');

    if (direct) return direct;

    const form = getComposerForm();

    if (form) {
      const buttons = [...form.querySelectorAll('button')];

      const byAttr = buttons.find(btn => {
        const s = [
          btn.id,
          btn.getAttribute('data-testid'),
          btn.getAttribute('aria-label'),
          btn.getAttribute('title'),
          btn.type,
          btn.className
        ].join(' ').toLowerCase();

        return (
          s.includes('send') ||
          s.includes('submit') ||
          s.includes('전송') ||
          s.includes('보내기') ||
          btn.type === 'submit'
        );
      });

      if (byAttr) return byAttr;
    }

    return null;
  }

  function getStopButton() {
    return document.querySelector('button[data-testid="stop-button"]')
      || document.querySelector('button[aria-label*="중지"]')
      || document.querySelector('button[aria-label*="Stop"]');
  }

  function getText(el) {
    if (!el) return '';
    if ('value' in el) return String(el.value || '').trim();
    return String(el.textContent || '').trim();
  }

  function isSendButtonReady(btn) {
    if (!btn) return false;
    if (btn.disabled) return false;
    if (btn.getAttribute('aria-disabled') === 'true') return false;
    return true;
  }

  /******************************************************************
   * ERROR DETECTION
   ******************************************************************/

  function getPageTailText() {
    const text = document.body?.innerText || '';
    return text.slice(-5000);
  }

  function getLatestTurnText() {
    const turns = [...document.querySelectorAll('[data-testid^="conversation-turn"]')]
      .filter(el => (el.innerText || el.textContent || '').trim());

    const latest = turns[turns.length - 1];
    return latest ? (latest.innerText || latest.textContent || '') : '';
  }

  function hasRetryableError() {
    const latest = getLatestTurnText();
    const tail = getPageTailText();
    const text = `${latest}\n${tail}`.toLowerCase();

    return (
      text.includes('메시지 전송 시간이 초과되었습니다') ||
      text.includes('다시 시도해 주세요') ||
      text.includes('다시 시도') ||
      text.includes('생각 중지됨') ||
      text.includes('생각 실패') ||
      text.includes('오류가 발생') ||
      text.includes('문제가 발생') ||
      text.includes('network error') ||
      text.includes('timed out') ||
      text.includes('timeout') ||
      text.includes('something went wrong') ||
      text.includes('try again')
    );
  }

  function recoverAndResend(reason) {
    recoveryCount += 1;

    awaitingMdDownload = false;
    harvesting = false;
    noMdRetries = 0;
    autoPaused = false;
    preExistingMdKeys = new Set();

    console.error('### E2R RECOVER AND RESEND ###', {
      reason,
      recoveryCount,
      delayMs: ERROR_RECOVERY_DELAY
    });

    if (scheduled) return;

    scheduled = true;

    setTimeout(() => {
      scheduled = false;
      sendNow();
    }, ERROR_RECOVERY_DELAY);
  }

  /******************************************************************
   * INPUT PROMPT
   ******************************************************************/

  function setText(el, text) {
    el.focus();

    if (el.isContentEditable) {
      el.textContent = '';

      const range = document.createRange();
      const selection = window.getSelection();

      range.selectNodeContents(el);
      range.collapse(false);
      selection.removeAllRanges();
      selection.addRange(range);

      document.execCommand('insertText', false, text);

      el.dispatchEvent(new InputEvent('input', {
        bubbles: true,
        cancelable: true,
        inputType: 'insertText',
        data: text
      }));

      el.dispatchEvent(new Event('change', { bubbles: true }));
      return;
    }

    if (el.tagName === 'TEXTAREA') {
      const setter = Object.getOwnPropertyDescriptor(
        HTMLTextAreaElement.prototype,
        'value'
      ).set;

      setter.call(el, '');

      el.dispatchEvent(new InputEvent('input', {
        bubbles: true,
        cancelable: true,
        inputType: 'deleteContentBackward',
        data: null
      }));

      let current = '';

      for (const ch of text) {
        const previous = current;
        current += ch;

        if (el._valueTracker) {
          el._valueTracker.setValue(previous);
        }

        el.dispatchEvent(new InputEvent('beforeinput', {
          bubbles: true,
          cancelable: true,
          inputType: 'insertText',
          data: ch
        }));

        setter.call(el, current);

        el.dispatchEvent(new InputEvent('input', {
          bubbles: true,
          cancelable: true,
          inputType: 'insertText',
          data: ch
        }));
      }

      el.dispatchEvent(new Event('change', { bubbles: true }));
    }
  }

  /******************************************************************
   * MD DOWNLOAD BUTTON LOGIC
   ******************************************************************/

  function getClickedSet() {
    try {
      return new Set(JSON.parse(localStorage.getItem(CLICKED_KEY) || '[]'));
    } catch {
      return new Set();
    }
  }

  function saveClickedSet(set) {
    localStorage.setItem(CLICKED_KEY, JSON.stringify([...set]));
  }

  function getConversationId() {
    const m = location.pathname.match(/\/c\/([^/?#]+)/);
    return m ? m[1] : 'no-conversation-id';
  }

  function getNearestTurnId(el) {
    const turn = el.closest('[data-testid^="conversation-turn"]');
    return turn?.getAttribute('data-testid') || 'no-turn';
  }

  function getButtonSignature(btn) {
    const text = (btn.innerText || btn.textContent || '').trim();
    const aria = btn.getAttribute?.('aria-label') || '';
    const title = btn.getAttribute?.('title') || '';
    const cls = String(btn.className || '');
    const turnId = getNearestTurnId(btn);

    return {
      text,
      aria,
      title,
      cls,
      turnId,
      haystack: `${text} ${aria} ${title} ${cls}`.toLowerCase()
    };
  }

  function isLikelyMdDownloadButton(btn) {
    const sig = getButtonSignature(btn);
    const { text, haystack, cls } = sig;

    if (!text) return false;
    if (text.length > 180) return false;

    const looksLikeEntityButton =
      cls.includes('behavior-btn') ||
      cls.includes('entity-underline');

    const hasDownloadWord =
      haystack.includes('다운로드') ||
      haystack.includes('download');

    const hasMdSignal =
      haystack.includes('.md') ||
      haystack.includes(' md ') ||
      haystack.includes('markdown') ||
      text.toLowerCase().endsWith('.md') ||
      text.toLowerCase().includes('.md');

    const filenameLikeMd =
      /^[^\n\r]{1,160}\.md$/i.test(text) ||
      /^[^\n\r]{1,160}\.md\s*다운로드$/i.test(text);

    return (
      filenameLikeMd ||
      (looksLikeEntityButton && hasDownloadWord) ||
      (hasDownloadWord && hasMdSignal)
    );
  }

  function getButtonKey(btn) {
    const conversationId = getConversationId();
    const sig = getButtonSignature(btn);

    return `${conversationId}::${sig.turnId}::${sig.text}`;
  }

  function getCandidateButtons() {
    return [...document.querySelectorAll('button')]
      .map((btn, domIndex) => {
        const sig = getButtonSignature(btn);
        const key = getButtonKey(btn);
        const rect = btn.getBoundingClientRect();

        return {
          btn,
          domIndex,
          key,
          text: sig.text,
          aria: sig.aria,
          title: sig.title,
          cls: sig.cls,
          turnId: sig.turnId,
          top: rect.top,
          bottom: rect.bottom
        };
      })
      .filter(x => isLikelyMdDownloadButton(x.btn));
  }

  function snapshotExistingMdKeys() {
    return new Set(getCandidateButtons().map(x => x.key));
  }

  function findLatestNewUnclickedMdButton() {
    const clicked = getClickedSet();

    const candidates = getCandidateButtons()
      .filter(x => !clicked.has(x.key))
      .filter(x => !preExistingMdKeys.has(x.key))
      .sort((a, b) => b.domIndex - a.domIndex);

    return candidates[0] || null;
  }

  function findLatestAnyUnclickedMdButton() {
    const clicked = getClickedSet();

    const candidates = getCandidateButtons()
      .filter(x => !clicked.has(x.key))
      .sort((a, b) => b.domIndex - a.domIndex);

    return candidates[0] || null;
  }

  function clickMdTarget(target, label) {
    if (!target) return false;

    console.log(`### E2R ${label} MD BUTTON TARGET ###`, {
      domIndex: target.domIndex,
      turnId: target.turnId,
      text: target.text,
      key: target.key
    });

    target.btn.scrollIntoView({ block: 'center' });
    target.btn.focus();
    target.btn.click();

    const clicked = getClickedSet();
    clicked.add(target.key);
    saveClickedSet(clicked);

    console.error('### E2R MD BUTTON CLICKED AND MARKED ###', target.key);

    return true;
  }

  function clickLatestNewMdButton() {
    const target = findLatestNewUnclickedMdButton();

    if (!target) {
      console.warn('### E2R NO NEW UNCLICKED RECENT MD BUTTON ###');
      return false;
    }

    return clickMdTarget(target, 'NEW RECENT');
  }

  function clickLatestAnyMdButton() {
    const target = findLatestAnyUnclickedMdButton();

    if (!target) {
      console.warn('### E2R NO ANY UNCLICKED RECENT MD BUTTON ###');
      return false;
    }

    return clickMdTarget(target, 'ANY RECENT');
  }

  /******************************************************************
   * SEND FLOW
   ******************************************************************/

  function markSent(timer, mode) {
    count += 1;
    scheduled = false;
    awaitingMdDownload = true;
    harvesting = false;
    noMdRetries = 0;

    if (timer) clearInterval(timer);

    console.error('### E2R SENT ###', {
      count,
      mode,
      flow: 'prompt -> wait md download'
    });
  }

  function tryRequestSubmitFallback() {
    const form = getComposerForm();

    if (!form) return false;

    try {
      if (typeof form.requestSubmit === 'function') {
        form.requestSubmit();
        return true;
      }

      form.dispatchEvent(new SubmitEvent('submit', {
        bubbles: true,
        cancelable: true
      }));

      return true;
    } catch (err) {
      console.warn('### E2R REQUEST SUBMIT FALLBACK FAILED ###', err);
      return false;
    }
  }

  function sendNow() {
    if (autoPaused) {
      scheduled = false;
      console.warn('### E2R AUTO PAUSED, SEND SKIP ###');
      return;
    }

    if (awaitingMdDownload) {
      scheduled = false;
      console.warn('### E2R WAITING MD DOWNLOAD, SEND SKIP ###');
      return;
    }

    const editor = getEditor();

    if (!editor) {
      scheduled = false;
      console.warn('### E2R NO EDITOR ###');
      return;
    }

    const currentText = getText(editor);

    if (currentText && currentText !== PROMPT) {
      scheduled = false;
      console.warn('### E2R EDITOR NOT EMPTY, SKIP ###', currentText.slice(0, 300));
      return;
    }

    if (!currentText) {
      console.warn('### E2R INSERT PROMPT ###');
      setText(editor, PROMPT);
    }

    preExistingMdKeys = snapshotExistingMdKeys();

    console.warn('### E2R PRE-EXISTING MD SNAPSHOT ###', {
      count: preExistingMdKeys.size
    });

    let tries = 0;
    let fallbackTried = false;

    const timer = setInterval(function () {
      tries += 1;

      const btn = getSendButton();

      if (isSendButtonReady(btn)) {
        btn.click();
        markSent(timer, 'button-click');
        return;
      }

      if (!fallbackTried && tries >= 10) {
        fallbackTried = true;

        console.warn('### E2R TRY REQUEST SUBMIT FALLBACK ###', {
          tries,
          button: btn,
          editorText: getText(getEditor()).slice(0, 120)
        });

        const ok = tryRequestSubmitFallback();

        if (ok) {
          markSent(timer, 'requestSubmit-fallback');
          return;
        }
      }

      if (tries >= 30) {
        scheduled = false;
        clearInterval(timer);

        console.warn('### E2R SEND BUTTON NOT READY ###', {
          button: btn,
          disabled: btn ? btn.disabled : null,
          ariaDisabled: btn ? btn.getAttribute('aria-disabled') : null,
          text: getText(getEditor()).slice(0, 300)
        });
      }
    }, 500);
  }

  function loop() {
    const editor = getEditor();

    if (!editor) return;
    if (autoPaused) return;
    if (getStopButton()) return;
    if (scheduled) return;
    if (harvesting) return;
    if (awaitingMdDownload) return;

    const currentText = getText(editor);

    if (currentText && currentText !== PROMPT) return;

    scheduled = true;

    const delay = count === 0 ? FIRST_DELAY : NEXT_DELAY;

    console.warn('### E2R SCHEDULED ###', {
      nextCount: count + 1,
      delayMs: delay,
      flow: count === 0 ? 'first prompt send' : 'after md download -> next prompt'
    });

    setTimeout(sendNow, delay);
  }

  /******************************************************************
   * HARVEST FLOW
   ******************************************************************/

  function harvestLoop() {
    if (autoPaused) return;
    if (!awaitingMdDownload) return;
    if (harvesting) return;

    if (getStopButton()) return;

    harvesting = true;

    const clicked = clickLatestNewMdButton();

    if (!clicked) {
      noMdRetries += 1;

      const retryableError = hasRetryableError();

      console.warn('### E2R WAITING NEW MD BUTTON ###', {
        retries: noMdRetries,
        limit: NO_MD_RETRY_LIMIT,
        retryableError
      });

      // 핵심 패치:
      // 생각 실패 / 메시지 전송 시간 초과 / 다시 시도 오류면 md가 안 생기는 게 정상.
      // 이때는 멈추지 말고 waiting 상태를 풀고 같은 PROMPT를 다시 보낸다.
      if (retryableError && noMdRetries >= ERROR_NO_MD_RETRY_LIMIT) {
        harvesting = false;
        recoverAndResend('retryable error detected while waiting md');
        return;
      }

      // md 버튼이 끝까지 안 나오면 GPT가 파일을 안 만들었거나 오류 응답으로 끝난 것.
      // 기존처럼 pause하지 말고 자동 재전송.
      if (noMdRetries >= NO_MD_RETRY_LIMIT) {
        harvesting = false;
        recoverAndResend('md button not found after retry limit');
        return;
      }

      harvesting = false;
      return;
    }

    noMdRetries = 0;

    console.warn('### E2R DOWNLOAD CLICKED, WAIT SETTLE ###', DOWNLOAD_SETTLE_DELAY);

    setTimeout(() => {
      awaitingMdDownload = false;
      harvesting = false;
      preExistingMdKeys = new Set();

      console.error('### E2R MD DOWNLOAD DONE -> NEXT PROMPT ALLOWED ###');
    }, DOWNLOAD_SETTLE_DELAY);
  }

  /******************************************************************
   * PUBLIC DEBUG COMMANDS
   ******************************************************************/

  window.E2R_SEND_NOW = sendNow;

  window.E2R_RECOVER_NOW = function () {
    recoverAndResend('manual recover');
  };

  window.E2R_HARVEST_NOW = function () {
    const ok = clickLatestAnyMdButton();
    console.log('E2R_HARVEST_NOW_RESULT:', ok);
    return ok;
  };

  window.E2R_HARVEST_NEW_NOW = function () {
    const ok = clickLatestNewMdButton();
    console.log('E2R_HARVEST_NEW_NOW_RESULT:', ok);
    return ok;
  };

  window.E2R_PAUSE = function () {
    autoPaused = true;
    console.warn('### E2R PAUSED ###');
  };

  window.E2R_RESUME = function () {
    autoPaused = false;
    scheduled = false;
    harvesting = false;
    awaitingMdDownload = false;
    noMdRetries = 0;
    preExistingMdKeys = new Set();

    console.warn('### E2R RESUMED ###');
  };

  window.E2R_RESET_ALL = function () {
    count = 0;
    scheduled = false;
    harvesting = false;
    awaitingMdDownload = false;
    noMdRetries = 0;
    autoPaused = false;
    recoveryCount = 0;
    preExistingMdKeys = new Set();

    localStorage.removeItem(CLICKED_KEY);

    console.warn('### E2R RESET ALL DONE ###');
  };

  window.E2R_CLEAR_MD_CLICK_HISTORY = function () {
    localStorage.removeItem(CLICKED_KEY);
    console.warn('### E2R MD CLICK HISTORY CLEARED ###');
  };

  window.E2R_DEBUG = function () {
    const editor = getEditor();
    const btn = getSendButton();
    const candidates = getCandidateButtons();
    const clicked = getClickedSet();

    return {
      version: '7.1-simple-error-recover',
      count,
      scheduled,
      awaitingMdDownload,
      harvesting,
      noMdRetries,
      autoPaused,
      recoveryCount,
      retryableError: hasRetryableError(),
      editor,
      editorTag: editor ? editor.tagName : null,
      editorText: getText(editor),
      composerForm: getComposerForm(),
      sendButton: btn,
      sendButtonDisabled: btn ? btn.disabled : null,
      sendButtonAriaDisabled: btn ? btn.getAttribute('aria-disabled') : null,
      stopButton: getStopButton(),
      preExistingMdKeys: [...preExistingMdKeys],
      mdCandidates: candidates.map(x => ({
        domIndex: x.domIndex,
        turnId: x.turnId,
        text: x.text,
        key: x.key,
        clicked: clicked.has(x.key),
        preExisting: preExistingMdKeys.has(x.key)
      }))
    };
  };

  /******************************************************************
   * START
   ******************************************************************/

  setInterval(loop, 3000);
  setInterval(harvestLoop, HARVEST_POLL_MS);
})();
