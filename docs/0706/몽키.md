const CSV_STATE_KEY = 'E2R_CURRENT_CSV';

  function getCsvStateStorage() {
    try {
      return window.sessionStorage || localStorage;
    } catch {
      return localStorage;
    }
  }

  function getSavedCsvState() {
    return getCsvStateStorage().getItem(CSV_STATE_KEY);
  }

  function setSavedCsvState(csv) {
    getCsvStateStorage().setItem(CSV_STATE_KEY, csv);
  }

  function removeSavedCsvState() {
    getCsvStateStorage().removeItem(CSV_STATE_KEY);
  }
/******************************************************************
   * CSV SEQUENCE
   ******************************************************************/

  function getCurrentCsv() {
    const saved = getSavedCsvState();

    if (saved && CSV_SEQUENCE.includes(saved)) return saved;

    return null;
  }

  function buildPrompt() {
    const csv = getCurrentCsv();

    if (!csv) {
      console.warn('### E2R CSV NOT SET: call E2R_SET_CSV("news_YYYYMMDD.csv") first ###');
      return null;
    }

    const next = BASE_PROMPT.replace(
      /이번 세션 시작 파일:\s*\nnews_(?:\d{8}|YYYYMMDD)\.csv/,
      '이번 세션 시작 파일:\n' + csv
    );

    if (next === BASE_PROMPT) {
      console.error('### E2R CSV PLACEHOLDER NOT REPLACED ###', csv);
    }

    return next;
  }

  function advanceCsvAfterSuccess() {
    const current = getCurrentCsv();

    if (!current) {
      console.warn('### E2R CSV NOT SET, ADVANCE SKIP ###');
      return;
    }

    const index = CSV_SEQUENCE.indexOf(current);

    if (index < 0) {
      console.error('### E2R CURRENT CSV NOT IN SEQUENCE ###', current);
      return;
    }

    if (index + 1 >= CSV_SEQUENCE.length) {
      console.error('### E2R CSV SEQUENCE FINISHED ###', current);
      return;
    }

    const next = CSV_SEQUENCE[index + 1];
    setSavedCsvState(next);

    console.error('### E2R ADVANCED CSV ###', {
      from: current,
      to: next,
      index: index + 1,
      total: CSV_SEQUENCE.length
    });
  }

    console.warn('### E2R CURRENT CSV SET ###', csv);
    return true;
  }

      message: current ? 'ready' : 'CSV not set. Run E2R_SET_CSV("news_YYYYMMDD.csv") in this tab.'

      csvState: e2rGetCsv(),
      markerIndex: index,
      snippet: index >= 0 ? prompt.slice(index, index + 80) : prompt.slice(0, 200)

  function setText(el, text) {
    el.focus();

    if (el.isContentEditable) {
      el.textContent = '';

      const range = document.createRange();
      const selection = window.getSelection();

      range.selectNodeContents(el);
      range.collapse(false);
      selection.removeAllRanges();
      selection.addRange(range);

      document.execCommand('insertText', false, text);

      el.dispatchEvent(new InputEvent('input', {
        bubbles: true,
        cancelable: true,
        inputType: 'insertText',
        data: text
      }));

      el.dispatchEvent(new Event('change', { bubbles: true }));
      return;
    }

    if (el.tagName === 'TEXTAREA') {
      const setter = Object.getOwnPropertyDescriptor(
        HTMLTextAreaElement.prototype,
        'value'
      ).set;

      setter.call(el, '');

      el.dispatchEvent(new InputEvent('input', {
        bubbles: true,
        cancelable: true,
        inputType: 'deleteContentBackward',
        data: null
      }));

      let current = '';

      for (const ch of text) {
        const previous = current;
        current += ch;

        if (el._valueTracker) {
          el._valueTracker.setValue(previous);
        }

        el.dispatchEvent(new InputEvent('beforeinput', {
          bubbles: true,
          cancelable: true,
          inputType: 'insertText',
          data: ch
        }));

        setter.call(el, current);

        el.dispatchEvent(new InputEvent('input', {
          bubbles: true,
          cancelable: true,
          inputType: 'insertText',
          data: ch
        }));
      }

      el.dispatchEvent(new Event('change', { bubbles: true }));
    }
  }

  function getConversationId() {
    const m = location.pathname.match(/\/c\/([^/?#]+)/);

  function getButtonSignature(btn) {
    const text = (btn.innerText || btn.textContent || '').trim();
    const aria = btn.getAttribute?.('aria-label') || '';
    const title = btn.getAttribute?.('title') || '';
    const cls = String(btn.className || '');
    const turnId = getNearestTurnId(btn);

    return {
      text,
      aria,
      title,
      cls,
      turnId,
      haystack: `${text} ${aria} ${title} ${cls}`.toLowerCase()
    };
  }

  function isLikelyMdDownloadButton(btn) {
    const sig = getButtonSignature(btn);
    const { text, haystack, cls } = sig;

    if (!text) return false;
    if (text.length > 180) return false;

    const looksLikeEntityButton =
      cls.includes('behavior-btn') ||
      cls.includes('entity-underline');

    const hasDownloadWord =
      haystack.includes('다운로드') ||
      haystack.includes('download');

    const hasMdSignal =
      haystack.includes('.md') ||
      haystack.includes(' md ') ||
      haystack.includes('markdown') ||
      text.toLowerCase().endsWith('.md') ||
      text.toLowerCase().includes('.md');

    const filenameLikeMd =
      /^[^\n\r]{1,160}\.md$/i.test(text) ||
      /^[^\n\r]{1,160}\.md\s*다운로드$/i.test(text);

    return (
      filenameLikeMd ||
      (looksLikeEntityButton && hasDownloadWord) ||
      (hasDownloadWord && hasMdSignal)
    );
  }

  function getButtonKey(btn) {
    const conversationId = getConversationId();
    const sig = getButtonSignature(btn);

    return `${conversationId}::${sig.turnId}::${sig.text}`;
  }

  function getCandidateButtons() {
    return [...document.querySelectorAll('button')]
      .map((btn, domIndex) => {
        const sig = getButtonSignature(btn);
        const key = getButtonKey(btn);
        const rect = btn.getBoundingClientRect();

        return {
          btn,
          domIndex,
          key,
          text: sig.text,
          aria: sig.aria,
          title: sig.title,
          cls: sig.cls,
          turnId: sig.turnId,
          top: rect.top,
          bottom: rect.bottom
        };
      })
      .filter(x => isLikelyMdDownloadButton(x.btn));
  }

  function snapshotExistingMdKeys() {
    return new Set(getCandidateButtons().map(x => x.key));
  }

  function findLatestNewUnclickedMdButton() {
    const clicked = getClickedSet();

    const candidates = getCandidateButtons()
      .filter(x => !clicked.has(x.key))
      .filter(x => !preExistingMdKeys.has(x.key))
      .sort((a, b) => b.domIndex - a.domIndex);

    return candidates[0] || null;
  }

  function markSent(timer, mode) {

    scheduled = false;
    awaitingMdDownload = true;

  function tryRequestSubmitFallback() {
    const form = getComposerForm();

    if (!form) return false;

    try {
      if (typeof form.requestSubmit === 'function') {
        form.requestSubmit();
        return true;
      }

      form.dispatchEvent(new SubmitEvent('submit', {
        bubbles: true,
        cancelable: true
      }));

      return true;
    } catch (err) {
      console.warn('### E2R REQUEST SUBMIT FALLBACK FAILED ###', err);
      return false;
    }
  }

  function sendNow() {
    if (autoPaused) {
      scheduled = false;
      console.warn('### E2R AUTO PAUSED, SEND SKIP ###');
      return;
    }

    if (awaitingMdDownload) {
      scheduled = false;
      console.warn('### E2R WAITING MD DOWNLOAD, SEND SKIP ###');
      return;
    }

    const editor = getEditor();

    if (!editor) {
      scheduled = false;
      console.warn('### E2R NO EDITOR ###');
      return;
    }

    const promptToSend = buildPrompt();

    if (!promptToSend) {
      scheduled = false;
      console.warn('### E2R WAITING FOR CSV START: use E2R_SET_CSV(...) ###');
      return;
    }

    const currentText = getText(editor);

    if (currentText && currentText !== promptToSend) {
      scheduled = false;
      console.warn('### E2R EDITOR NOT EMPTY, SKIP ###', currentText.slice(0, 300));
      return;
    }

    if (!currentText) {
      console.warn('### E2R INSERT PROMPT ###', getCurrentCsv());
      setText(editor, promptToSend);
    }

    preExistingMdKeys = snapshotExistingMdKeys();

    console.warn('### E2R PRE-EXISTING MD SNAPSHOT ###', {
      count: preExistingMdKeys.size
    });

    let tries = 0;
    let fallbackTried = false;

    const timer = setInterval(function () {
      tries += 1;

      const btn = getSendButton();

      if (isSendButtonReady(btn)) {
        btn.click();
        markSent(timer, 'button-click');
        return;
      }

      if (!fallbackTried && tries >= 10) {
        fallbackTried = true;

        console.warn('### E2R TRY REQUEST SUBMIT FALLBACK ###', {
          tries,
          button: btn,
          editorText: getText(getEditor()).slice(0, 120)
        });

        const ok = tryRequestSubmitFallback();

        if (ok) {
          markSent(timer, 'requestSubmit-fallback');
          return;
        }
      }

      if (tries >= 30) {
        scheduled = false;
        clearInterval(timer);

        console.warn('### E2R SEND BUTTON NOT READY ###', {
          button: btn,
          disabled: btn ? btn.disabled : null,
          ariaDisabled: btn ? btn.getAttribute('aria-disabled') : null,
          text: getText(getEditor()).slice(0, 300)
        });
      }
    }, 500);
  }

  function loop() {
    const editor = getEditor();

    if (!editor) return;
    if (autoPaused) return;
    if (getStopButton()) return;
    if (scheduled) return;
    if (harvesting) return;
    if (awaitingMdDownload) return;

    const currentText = getText(editor);

    const promptToSend = buildPrompt();

    if (!promptToSend) return;
    if (currentText && currentText !== promptToSend) return;

    scheduled = true;

    const delay = count === 0 ? FIRST_DELAY : NEXT_DELAY;

    console.warn('### E2R SCHEDULED ###', {
      nextCount: count + 1,
      delayMs: delay,
      flow: count === 0 ? 'first prompt send' : 'after md download -> next prompt'
    });

    setTimeout(sendNow, delay);
  }

  /******************************************************************
   * HARVEST FLOW
   ******************************************************************/

  function harvestLoop() {
    if (autoPaused) return;
    if (!awaitingMdDownload) return;
    if (harvesting) return;

    if (getStopButton()) return;

    harvesting = true;

    const clicked = clickLatestNewMdButton();

    if (!clicked) {
      noMdRetries += 1;

      const retryableError = hasRetryableError();

      console.warn('### E2R WAITING NEW MD BUTTON ###', {
        retries: noMdRetries,
        limit: NO_MD_RETRY_LIMIT,
        retryableError
      });

      // 핵심 패치:
      // 생각 실패 / 메시지 전송 시간 초과 / 다시 시도 오류면 md가 안 생기는 게 정상.
      // 이때는 멈추지 말고 waiting 상태를 풀고 같은 PROMPT를 다시 보낸다.
      if (retryableError && noMdRetries >= ERROR_NO_MD_RETRY_LIMIT) {
        harvesting = false;
        recoverAndResend('retryable error detected while waiting md');
        return;
      }

      // md 버튼이 끝까지 안 나오면 GPT가 파일을 안 만들었거나 오류 응답으로 끝난 것.
      // 기존처럼 pause하지 말고 자동 재전송.
      if (noMdRetries >= NO_MD_RETRY_LIMIT) {
        harvesting = false;
        recoverAndResend('md button not found after retry limit');
        return;
      }

      harvesting = false;
      return;
    }

    noMdRetries = 0;

    console.warn('### E2R DOWNLOAD CLICKED, WAIT SETTLE ###', DOWNLOAD_SETTLE_DELAY);

    setTimeout(() => {
      advanceCsvAfterSuccess();

      awaitingMdDownload = false;
      harvesting = false;
      preExistingMdKeys = new Set();

      console.error('### E2R MD DOWNLOAD DONE -> NEXT PROMPT ALLOWED ###');
    }, DOWNLOAD_SETTLE_DELAY);
  }

      }

      window.E2R_SET_CSV = function (csv) {
        if (!csvSequence.includes(csv)) {
          console.error('### E2R CSV NOT IN SEQUENCE ###', csv);
          return false;
        }

        csvStorage().setItem(csvStateKey, csv);
        console.warn('### E2R CURRENT CSV SET ###', csv);
        return true;
      };

      window.E2R_GET_CSV = function () {
        let current = csvStorage().getItem(csvStateKey);

        if (!current || !csvSequence.includes(current)) current = null;

        const index = csvSequence.indexOf(current);

        return {
          current,
          index,
          ready: Boolean(current),
          total: csvSequence.length,
          next: index >= 0 ? csvSequence[index + 1] || null : null,
          message: current ? 'ready' : 'CSV not set. Run E2R_SET_CSV("news_YYYYMMDD.csv") in this tab.'
        };
      };

      window.E2R_ADVANCE_CSV = function () {
        const state = window.E2R_GET_CSV();

        if (!state.next) {
          console.error('### E2R CSV SEQUENCE FINISHED ###', state.current);
          return state;
        }

        csvStorage().setItem(csvStateKey, state.next);
        console.warn('### E2R ADVANCED CSV ###', {
          from: state.current,
          to: state.next,
          index: state.index + 1,
          total: state.total
        });

        return window.E2R_GET_CSV();
      };

      window.E2R_PREVIEW_PROMPT = function () {
        const state = window.E2R_GET_CSV();

        return {
          csvState: state,
          note: '이 bridge preview는 CSV 진행 상태만 보여준다. 실제 전송 프롬프트는 Tampermonkey 내부 buildPrompt가 같은 tab sessionStorage 값을 사용한다.'
        };
      };

    } + ')(' + JSON.stringify(CSV_SEQUENCE) + ', ' + JSON.stringify(CSV_STATE_KEY) + ');';

    try {

    } catch (err) {

    }
